import cmd
import subprocess
def call_tree_diagram(target, depth=20, diagram='sequence'):
    assert diagram in ['sequence', 'flow', 'ascii']
    project_root = '../../..'
    if diagram == 'ascii':
        diagram_opt = ''
    else:
        diagram_opt = f'--diagram {diagram}'
    cmd = f'python {project_root}/scripts/call_tree_analyzer.py {project_root}/src/phasic --quiet --callable {target} {diagram_opt} -d {depth}'.split()
    output = subprocess.check_output(cmd).decode()
    if diagram == 'ascii':
        print(f"```{txt}\n{output}\n```")
    else:    
        label = 'fig-diagram-' + target.replace('.', '-')
        print(f"```{mermaid}\n%%| echo: false\n%%| label: {label}\n{output}\n```")
