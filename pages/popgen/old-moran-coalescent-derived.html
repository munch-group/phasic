def expected_bw_loss_time(N, n_samples, n_derived, s):
    # # --- parameters ---
    # N = 100
    # n_ancestral = 1
    # n_derived   = 1
    # n_samples   = n_ancestral + n_derived
    # s = 0.2
    gamma = 2 * N * s                      # scaled selection (haploid: 2Ns)

    # --- posterior over current/starting frequency (uniform prior) ---
    k = np.arange(1, N)
    p = k / N
    w = beta(1 + n_derived, 1 + (n_samples - n_derived)).pdf(p)
    w /= w.sum()

    # --- stable allele-frequency helpers ---
    def u_and_om(x, gamma):
        """u(x) and 1-u(x), stable for large |gamma| (no catastrophic cancellation)."""
        x = np.asarray(x, float)
        if abs(gamma) < 1e-12:
            return x.copy(), 1.0 - x
        denom = np.expm1(-gamma)                                  # e^{-g}-1  (<0)
        u  = np.expm1(-gamma * x) / denom
        om = np.exp(-gamma * x) * np.expm1(-gamma * (1 - x)) / denom   # = 1-u, factored
        return u, om

    def uprime(x, gamma):
        x = np.asarray(x, float)
        if abs(gamma) < 1e-12:
            return np.ones_like(x)
        return -gamma * np.exp(-gamma * x) / np.expm1(-gamma)     # u'(x) > 0

    # --- (1-u(p)) * t0(p) = loss-weighted sojourn integral (no unstable division) ---
    def loss_weighted_sojourn(p_val, N, gamma, n_grid=4000):
        x = np.arange(1, n_grid) / n_grid
        u, om   = u_and_om(x, gamma)
        up, omp = (v[0] for v in u_and_om(np.array([p_val]), gamma))
        V   = x * (1 - x) / N
        upr = uprime(x, gamma)
        integ = (2.0 / (V * upr)) * np.where(x > p_val, up * om**2, u * omp * om)
        return np.trapezoid(integ, x), omp        # (1-u(p))*t0(p) ,  and 1-u(p)

    # --- evaluate on posterior grid ---
    num   = np.empty_like(p)                  # (1-u(p)) * t0(p)
    omp_p = np.empty_like(p)                  # 1 - u(p)  (loss prob)
    for i, pi in enumerate(p):
        num[i], omp_p[i] = loss_weighted_sojourn(pi, N, gamma)

    # average over posterior AND condition on loss
    E_loss = np.sum(w * num) / np.sum(w * omp_p)
    return E_loss
def coalescent_with_derived(state, indexer=None, N=None):
    transitions = []

    # callback params are always float
    N = int(N)

    # derived and total live lineages
    n_anc, n_der = 0, 0
    for i in range(indexer.lineages.state_length):
        p = indexer.lineages.i2p(i)
        if p.is_derived:
            n_der += state[i]
        else:
            n_anc += state[i]
    n_live = n_der + n_anc

    if n_live == 1:
        # absorbing state
        return []

    # pop derived freq
    freq = int(state[indexer.derived_freq])

    for i, j in all_pairs(range(indexer.lineages.state_length)):

        # props of lineage types at index i and j
        pi = indexer.lineages.i2p(i)
        pj = indexer.lineages.i2p(j)

        # both either anc or der
        same_ancestry = bool(pi.is_derived) == bool(pj.is_derived) or freq == 0

        # only coalesce derived and ancestral if derived is lost from population (implies the derived is now ancestral too)
        if not same_ancestry and freq > 0:
        # if not same_ancestry and freq > 1:
            continue

        same = int(i == j)
        if same and state[i] < 2:
            continue
        if not same and (state[i] < 1 or state[j] < 1):
            continue
        new = state.copy()
        new[i] -= 1
        new[j] -= 1
        is_derived = pi.is_derived if same_ancestry else 0 # can only be 0 if this is the final coalescence
        k = indexer.lineages.p2i(descendants=pi.descendants+pj.descendants, is_derived=is_derived)
        new[k] += 1

        pair_count = state[i] * (state[j] - same) / (1 + same)
        assert pair_count >= 1
        # if pi.is_derived and pj.is_derived:
        #     norm_freq = N / freq
        # else:
        #     # norm_freq = 1
        #     norm_freq = N / (N-freq)
        # coef = [pair_count * norm_freq, 0, 0] 

        if pi.is_derived and pj.is_derived:
            allele_N = freq
            coef = [pair_count, 0, 0, allele_N] 
        else:
            # norm_freq = 1
            allele_N = N-freq
            coef = [pair_count, 0, allele_N,0] 
        # coef = [pair_count, allele_N, 0, 0] 

        transitions.append([new, coef])

    # transitions to sister states with other derived population frequencies
    # if n_live > 1 and 0 < freq < N:
    if n_live > 1 and n_der <= freq < N:

        # freq increase backwards in time (forwards death derived, birth ancestral)
        f = freq + 1    
        if freq != 0 and f < N:# or f==N and n_der > 0):
            new = state.copy()
            new[indexer.derived_freq] = f
            coef = [0, 0, freq, 0]
            assert freq
            transitions.append([new, coef])

        # freq decrease backwards in time (s >= 0) (forwards death ancestral, birth derived)
        # f = freq - 1 # pos sel *reduces* freq backwards in time
        # if f >= n_der:
        if f > 0 or (f == 0 and n_der == 1):
            new = state.copy()
            new[indexer.derived_freq] = f
            # coef = [0, 0, 0, freq - 1] # freq-1 because the ancestral lineage that dies could be the one that just switched to derived in the forward direction, so freq only goes down by 1 if freq>1
            coef = [0, 0, 0, freq]
            assert freq
            transitions.append([new, coef])

    return transitions


def weight_update(param, coef):
    N, s = param
    coal_pairs, allele_N, freq_up, freq_down = coef
    if coal_pairs > 0:
        # rate = coal_pairs * 2 / allele_N**2

        # allele_N = max(1, allele_N)
        # allele_N = min(N-1, allele_N)

#        rate = coal_pairs * 2 / (allele_N * (N - allele_N))


        assert bool(freq_up) != bool(freq_down) or freq_up == freq_down, "only one of freq_up and freq_down should be nonzero, unless they are both zero (in which case this is a coalescence transition and not a frequency change transition)"
        if freq_up:
            freq = freq_up

            # freq = min(N-1, freq+1)

            der_death = freq / (N + freq*s)
            anc_birth = (N - freq) / N
            rate = der_death * anc_birth * coal_pairs / (((N - freq)*(N - freq-1)/2) + ((freq)*(freq-1)/2) )
        else:
            freq = freq_down

            # freq = max(1, freq-1)

            anc_death = (N - freq) / N
            der_birth = (1 + s) * freq / (N + s * freq)
            rate = anc_death * der_birth * coal_pairs / (((N - freq)*(N - freq-1)/2) + ((freq)*(freq-1)/2) )


    elif freq_up: # freq down forwards in time
        freq = freq_up

#        freq = min(N-1, freq+1)

        der_death = freq / (N + freq*s)
        anc_birth = (N - freq) / N
        rate = der_death * anc_birth        
        # rate = freq * (N - freq) / (N * (N + s*freq))
    elif freq_down: # freq up forwards in time
        _s = - s # s is negative because time is backwards
        freq = freq_down

#        freq = max(1, freq-1)

        anc_death = (N - freq) / N
        der_birth = (1 + s) * freq / (N + s * freq)
        rate = anc_death * der_birth
        # rate = (1 + s) * freq * (N - freq) / (N * (N+s*freq))
    else:
        assert 0
    return rate

def simulate_loss_times(N, s, p0_draws, rng, max_gen=10_000_000):
    """Forward haploid Wright-Fisher with selection (derived fitness 1+s).
    Runs each replicate to absorption (vectorized over replicates).
    Returns (times, outcome): outcome 0 = derived lost, 1 = derived fixed."""
    counts  = np.clip(np.rint(p0_draws * N).astype(np.int64), 1, N - 1)
    t       = np.zeros(counts.shape, dtype=np.int64)
    outcome = np.full(counts.shape, -1, dtype=np.int8)   # -1 = still segregating
    active  = np.ones(counts.shape, dtype=bool)

    for _ in range(max_gen):
        idx = np.flatnonzero(active)
        if idx.size == 0:
            break
        p     = counts[idx] / N
        p_sel = p * (1 + s) / (1 + s * p)                # selection on derived
        new   = rng.binomial(N, p_sel)                   # binomial resampling
        counts[idx] = new
        t[idx] += 1
        lost, fixed = (new == 0), (new == N)
        outcome[idx[lost]]  = 0
        outcome[idx[fixed]] = 1
        active[idx[lost | fixed]] = False

    return t, outcome

# --- fixation probability under selection (Kimura) ---
def u_fix(x, gamma):
    x = np.asarray(x, dtype=float)
    if abs(gamma) < 1e-12:
        return x.copy()
    return (1.0 - np.exp(-gamma * x)) / (1.0 - np.exp(-gamma))

# --- conditional mean loss time t0(p), haploid N, via Green's function ---
def cond_loss_time(p_val, N, gamma, n_grid=4000):
    x   = np.arange(1, n_grid) / n_grid           # interior (0,1)
    u   = u_fix(x, gamma)
    up  = u_fix(np.array([p_val]), gamma)[0]
    V   = x * (1 - x) / N
    psi = np.exp(-gamma * x)
    I   = (1 - np.exp(-gamma)) / gamma if abs(gamma) > 1e-12 else 1.0
    G = np.where(x > p_val,
                 2 * I * up * (1 - u) / (V * psi),    # start below x
                 2 * I * u  * (1 - up) / (V * psi))   # start above x
    return np.trapezoid(G * (1 - u), x) / (1 - up)

# average over posterior AND condition on loss
E_loss = np.sum(w * loss_prob * t0) / np.sum(w * loss_prob)

def sample_joint_observations(joint_graph, theta, nr_observations):
    joint_graph.update_weights(theta + [mutation_rate], callback=weight_update) 
    joint_prob_table = joint_graph.joint_prob_table()
    p = joint_prob_table['prob'] / joint_prob_table['prob'].sum()
    p = p.to_numpy()
    sample = np.random.choice(joint_prob_table.index.values, nr_observations, p=p)

def sample_joint_observations(joint_prob_graph, theta, nr_observations):
    joint_prob_graph.update_weights(theta) 
    joint_prob_table = joint_prob_graph.joint_prob_table()
    p = joint_prob_table['prob'] / joint_prob_table['prob'].sum()
    p = p.to_numpy()
    sample = np.random.choice(joint_prob_table.index.values, nr_observations, p=p)
    observations = joint_prob_table.loc[sample, joint_prob_table.columns[:-1]].to_numpy().tolist()
    return observations
